Google Spreadsheet Live Sync

Record check-ins automatically into your Google Sheet

Every check-in will send Organization, Name, Email, Timestamp, and Signature data to this endpoint.

How to connect your Google Sheet (1 minute):

  1. Open your Google Sheet → Click Extensions > Apps Script.
  2. Paste the 5-line script below into the editor.
  3. Click Deploy > New deployment > Select type: Web app.
  4. Set "Who has access" to Anyone, deploy, and paste the URL above!
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([new Date(), data.organization, data.name, data.email, data.status, data.signature]);
  return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
}
